Demo 1 - 一行引入 autoload.js

最简单的方式。autoload.js 会自己加载样式、引擎、模型数据。

<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>我的网站</title>
</head>
<body>

    <h1>网站内容</h1>

    <!-- 引入看板娘，就这一行 -->
    <script src="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/autoload.js"></script>
</body>
</html>

想改默认参数，在引入之前定义一个全局对象：

<script>
    window.live2d_autoload = {
        modelId: 3,              // 默认角色
        modelTexturesId: 0,      // 默认皮肤，0 = 原皮
        waifuMinWidth: '768px',  // 页面窄于此宽度时隐藏
        waifuEdgeSide: 'right:0' // 停靠位置
    };
</script>
<script src="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/autoload.js"></script>

Demo 2 - 手动控制加载

需要自己决定加载时机、或者要改更多参数时用这种方式。顺序不能乱。

<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>我的网站</title>

    <!-- 1. 样式：element-ui 图标 + 看板娘样式 -->
    <link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/element-ui@2.15.14/lib/theme-chalk/icon.css">
    <link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/waifu.css">
</head>
<body>

    <!-- 2. 看板娘容器（结构固定，顺序有讲究） -->
    <div class="waifu">
        <div class="waifu-tips"></div>
        <canvas id="live2d" class="live2d" width="280" height="250"></canvas>
        <div class="waifu-tool">
            <span class="el-icon-house"></span>
            <span class="el-icon-chat-dot-round"></span>
            <span class="el-icon-user"></span>
            <span class="el-icon-magic-stick"></span>
            <span class="el-icon-camera"></span>
            <span class="el-icon-view"></span>
            <span class="el-icon-discover"></span>
            <span class="el-icon-switch-button"></span>
        </div>
    </div>

    <!-- 3. 依赖与引擎：引擎 → 静态 API → 控制逻辑，顺序不能反 -->
    <script src="https://cdn.jsdelivr.net/npm/jquery@3.3.1/dist/jquery.min.js"></script>
    <script src="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/live2d.js"></script>
    <script src="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/live2d-static-api.js"></script>
    <script src="https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/waifu-tips.js"></script>

    <!-- 4. 装静态 API，然后启动 -->
    <script>
        var MODELS = 'https://cdn.jsdelivr.net/gh/whyyy-404/live2d-models@main';

        fetch(MODELS + '/models.json')
            .then(function (r) { return r.json(); })
            .then(function (data) {
                var api = createStaticModelAPI({
                    base: MODELS + '/',
                    data: data
                });
                api.install();

                live2d_settings['modelId'] = 1;
                live2d_settings['modelTexturesId'] = 0;
                live2d_settings['waifuEdgeSide'] = 'right:0';
                live2d_settings['modelAPI'] = MODELS + '/api/';

                initModel('https://cdn.jsdelivr.net/gh/whyyy-404/live2d-waifu@main/assets/waifu-tips.json');
            });
    </script>
</body>
</html>

为什么需要 api.install()：库本来是对着 fghrsh 的 PHP 后端写的，会请求 get/、switch/ 这些接口。live2d-static-api.js 把这些请求在浏览器里就地应答掉，所以不需要任何后端。

Demo 3 - 常用配置项

全部写在 initModel() 之前。完整清单见 README。

/* 模型 */
live2d_settings['modelId']             = 1;           // 1 Pio / 2 Tia / 3 22娘 / 4 33娘 / 5 雫 / 6 海王星 / 7 舰娘
live2d_settings['modelTexturesId']     = 0;           // 默认皮肤，0 = 原皮
live2d_settings['modelStorage']        = true;        // 记住用户选的角色与皮肤

/* 位置与尺寸 */
live2d_settings['waifuEdgeSide']       = 'right:0';   // 停靠位置：'right:0' 或 'left:0'
live2d_settings['waifuMinWidth']       = '768px';     // 页面窄于此宽度时隐藏，'disable' = 永不隐藏
live2d_settings['waifuToolTop']        = '30px';      // 工具栏距顶部的距离

/* 拖拽 */
live2d_settings['waifuDraggable']      = 'axis-x';    // 'axis-x'（吸附边缘）或 'disable'
live2d_settings['waifuDraggableSnap']  = 1000;        // 吸附动画时长（毫秒）

/* 一言 */
live2d_settings['showHitokoto']        = true;        // 是否显示一言按钮
live2d_settings['hitokotoInterval']    = 0;           // 空闲自动换一言的间隔，0 = 只手动触发

/* 工具栏按钮开关 */
live2d_settings['canCloseLive2d']      = true;        // 关闭
live2d_settings['canSwitchModel']      = true;        // 换角色
live2d_settings['canSwitchTextures']   = true;        // 换皮肤
live2d_settings['canTakeScreenshot']   = true;        // 拍照
live2d_settings['canTurnToHomePage']   = true;        // 首页
live2d_settings['canTurnToAboutPage']  = true;        // 关于
live2d_settings['canHideWaifu']        = true;        // 悬停隐藏

换掉默认存储：默认用 localStorage（按网站分别记忆）。想跨网站同步，覆盖 storage 即可：

live2d_settings['storage'] = {
    save: function (key, value) { GM_setValue(key, value); },
    load: function (key) { return GM_getValue(key, null); }
};

Demo 4 - 自定义台词

台词全部放在一个 JSON 里。下载下来改，或者用自己的地址替换掉 initModel() 的参数。

// 用你自己的台词文件
initModel('https://your-site.com/my-waifu-tips.json');

文件结构：

{
    "waifu": {
        "hour_tips": {
            "t7-11": ["上午好~", "今天也要加油"]
        },
        "model_message": {
            "1": ["来自 Potion Maker 的 Pio 酱", "药水铺的配方还没调完呢"]
        }
    },
    "mouseover": [
        { "selector": ".waifu-tool .el-icon-user", "text": ["要换角色吗？"] }
    ],
    "click": [
        { "selector": ".waifu #live2d", "text": ["别戳我啦"] }
    ],
    "seasons": [
        { "date": "01/01", "text": ["元旦快乐"] }
    ]
}

seasons 的 date 只支持公历，格式 MM/DD 或 MM/DD-MM/DD。文本里的 {year} 会替换成当前年份。

现在这份台词库：waifu-tips.json（通用版，31 个节日 / 7 个模型招呼语 / 15 种来源识别）
另有一份 waifu-tips-blog.json，保留了 fghrsh 原版给博客主题用的悬停台词。
